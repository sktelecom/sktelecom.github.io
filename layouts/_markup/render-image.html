{{- /* 이미지 렌더 훅 — 번들 PNG/JPEG는 최대 폭 제한 + WebP 변환으로 전송량을 줄이고, 치수·디코딩·로딩 힌트로 CLS/LCP를 개선한다. 첫 이미지(.Ordinal 0)는 LCP 후보로 보고 eager+high, 나머지는 lazy. SVG·GIF(애니메이션 보존)·외부 URL은 원본을 그대로 쓴다. */ -}} {{- $dest := .Destination -}} {{- $img := "" -}} {{- if not (strings.HasPrefix $dest "http") -}} {{- $path := strings.TrimPrefix "./" $dest -}} {{- with .Page.Resources.GetMatch $path -}}{{- $img = . -}}{{- end -}} {{- if not $img -}}{{- with resources.Get $path -}}{{- $img = . -}}{{- end -}}{{- end -}} {{- end -}} {{- $eager := eq .Ordinal 0 -}} {{- $maxWidth := 1600 -}} {{- $src := $dest -}} {{- $width := "" -}} {{- $height := "" -}} {{- with $img -}} {{- $subType := .MediaType.SubType -}} {{- if or (eq $subType "png") (eq $subType "jpeg") -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- $targetW := $maxWidth -}} {{- if lt .Width $maxWidth -}}{{- $targetW = .Width -}}{{- end -}} {{- $resized := .Resize (printf "%dx webp q82" $targetW) -}} {{- $src = $resized.RelPermalink -}} {{- else if and (ne $subType "svg") .Width .Height -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- end -}} {{- end -}}
